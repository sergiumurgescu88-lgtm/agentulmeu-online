# ============================================
# AgentulMeu Flask Backend — Systemd Service
# Salvează ca: /etc/systemd/system/agentulmeu-flask.service
# ============================================

[Unit]
Description=AgentulMeu.online Flask Backend Server
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/var/www/agentulmeu.online
Environment="PATH=/root/.local/bin:/usr/local/bin:/usr/bin:/bin"
Environment="PYTHONIOENCODING=utf-8"
Environment="PORT=8080"
Environment="HOST=127.0.0.1"
Environment="FLASK_DEBUG=false"
Environment="HERMES_CMD=hermes"
Environment="OLLAMA_URL=http://localhost:11434"
ExecStart=/usr/bin/python3 /var/www/agentulmeu.online/server.py
Restart=always
RestartSec=5
StandardOutput=append:/var/log/agentulmeu-flask.log
StandardError=append:/var/log/agentulmeu-flask.log

[Install]
WantedBy=multi-user.target
